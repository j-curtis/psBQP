\section*{Crank-Nicolson Discretization of Usadel Equation}

\subsection*{Simple Example: Two Derivative Terms}

Consider the equation:
\begin{equation}
i\hat{\tau}_3 \frac{\partial g}{\partial t_1} + \frac{\partial g}{\partial t_2} i\hat{\tau}_3 = 0,
\end{equation}
where $g = g(t_1, t_2)$ is a function of two time arguments.

\paragraph{Goal:} Evolve $g$ from time $t$ to $t+\Delta t$ in the first argument $t_1$, holding $t_2 = t'$ fixed.

\paragraph{Step 1: Integrate over time interval.}

Integrate the equation from $t$ to $t+\Delta t$ in the $t_1$ direction:
\begin{equation}
\int_t^{t+\Delta t} dt_1 \left[i\hat{\tau}_3 \frac{\partial g(t_1, t')}{\partial t_1} + \frac{\partial g(t_1, t')}{\partial t'} i\hat{\tau}_3\right] = 0.
\end{equation}

\paragraph{Step 2: Forward Euler (explicit scheme).}

Evaluate the integrand at the \emph{beginning} of the interval ($t_1 = t$):
\begin{equation}
\Delta t \cdot \left[i\hat{\tau}_3 \frac{\partial g(t, t')}{\partial t_1}\bigg|_{t_1=t} + \frac{\partial g(t, t')}{\partial t'} i\hat{\tau}_3\right] = 0.
\end{equation}

Using finite differences:
\begin{align}
\frac{\partial g}{\partial t_1}\bigg|_{t_1=t} &\approx \frac{g(t+\Delta t, t') - g(t, t')}{\Delta t}, \\
\frac{\partial g}{\partial t'} &\approx \frac{g(t, t'+\Delta t) - g(t, t')}{\Delta t}.
\end{align}

Substituting:
\begin{equation}
i\hat{\tau}_3 [g(t+\Delta t, t') - g(t, t')] + [g(t, t'+\Delta t) - g(t, t')] i\hat{\tau}_3 = 0.
\end{equation}

Solving for the unknown $g(t+\Delta t, t')$:
\begin{equation}
\boxed{i\hat{\tau}_3 \, g(t+\Delta t, t') = -[g(t, t'+\Delta t) - g(t, t')] i\hat{\tau}_3 + i\hat{\tau}_3 \, g(t, t').}
\end{equation}

This is an \textbf{explicit update}: $g(t+\Delta t, t')$ can be directly computed from known values at time $t$.

\paragraph{Step 3: Crank-Nicolson (implicit scheme).}

Evaluate the integrand at \emph{both} the beginning ($t_1 = t$) and end ($t_1 = t+\Delta t$) of the interval, then average:
\begin{equation}
\begin{split}
\frac{\Delta t}{2} &\left[i\hat{\tau}_3 \frac{\partial g(t, t')}{\partial t_1}\bigg|_{t_1=t} + \frac{\partial g(t, t')}{\partial t'} i\hat{\tau}_3\right] \\
+ \frac{\Delta t}{2} &\left[i\hat{\tau}_3 \frac{\partial g(t+\Delta t, t')}{\partial t_1}\bigg|_{t_1=t+\Delta t} + \frac{\partial g(t+\Delta t, t')}{\partial t'} i\hat{\tau}_3\right] = 0.
\end{split}
\end{equation}

Using finite differences:
\begin{align}
\frac{\partial g}{\partial t_1}\bigg|_{t_1=t} &\approx \frac{g(t+\Delta t, t') - g(t, t')}{\Delta t}, \\
\frac{\partial g(t, t')}{\partial t'} &\approx \frac{g(t, t'+\Delta t) - g(t, t')}{\Delta t}, \\
\frac{\partial g}{\partial t_1}\bigg|_{t_1=t+\Delta t} &\approx \frac{g(t+2\Delta t, t') - g(t+\Delta t, t')}{\Delta t}, \\
\frac{\partial g(t+\Delta t, t')}{\partial t'} &\approx \frac{g(t+\Delta t, t'+\Delta t) - g(t+\Delta t, t')}{\Delta t}.
\end{align}

Substituting and simplifying:
\begin{equation}
\begin{split}
&\frac{1}{2}\left[i\hat{\tau}_3 (g(t+\Delta t, t') - g(t, t')) + (g(t, t'+\Delta t) - g(t, t')) i\hat{\tau}_3\right] \\
&+ \frac{1}{2}\left[i\hat{\tau}_3 (g(t+2\Delta t, t') - g(t+\Delta t, t')) + (g(t+\Delta t, t'+\Delta t) - g(t+\Delta t, t')) i\hat{\tau}_3\right] = 0.
\end{split}
\end{equation}

\paragraph{Problem:} This involves $g(t+2\Delta t, t')$, which we don't know yet!

\paragraph{Modified Crank-Nicolson:} Use centered difference for $\partial_{t_1}$ derivative only:
\begin{equation}
\frac{\partial g}{\partial t_1}\bigg|_{t_1=t+\Delta t/2} \approx \frac{g(t+\Delta t, t') - g(t, t')}{\Delta t},
\end{equation}
and average the $\partial_{t'}$ derivative at times $t$ and $t+\Delta t$:
\begin{equation}
\frac{1}{2}\left[\frac{\partial g(t, t')}{\partial t'} + \frac{\partial g(t+\Delta t, t')}{\partial t'}\right] \approx \frac{1}{2}\left[\frac{g(t, t'+\Delta t) - g(t, t')}{\Delta t} + \frac{g(t+\Delta t, t'+\Delta t) - g(t+\Delta t, t')}{\Delta t}\right].
\end{equation}

The Crank-Nicolson discretization becomes:
\begin{equation}
\begin{split}
i\hat{\tau}_3 &[g(t+\Delta t, t') - g(t, t')] \\
&+ \frac{1}{2}\left[(g(t, t'+\Delta t) - g(t, t')) + (g(t+\Delta t, t'+\Delta t) - g(t+\Delta t, t'))\right] i\hat{\tau}_3 = 0.
\end{split}
\end{equation}

Rearranging to separate knowns (time $t$) from unknowns (time $t+\Delta t$):
\begin{equation}
\boxed{
\begin{split}
&i\hat{\tau}_3 \, g(t+\Delta t, t') + \frac{1}{2}(g(t+\Delta t, t'+\Delta t) - g(t+\Delta t, t')) i\hat{\tau}_3 \\
&= i\hat{\tau}_3 \, g(t, t') - \frac{1}{2}(g(t, t'+\Delta t) - g(t, t')) i\hat{\tau}_3.
\end{split}
}
\end{equation}

\paragraph{Key observation:} The LHS contains both $g(t+\Delta t, t')$ and $g(t+\Delta t, t'+\Delta t)$. This creates an \textbf{implicit system} where all values at the new time $t+\Delta t$ are coupled and must be solved simultaneously.

\subsection*{Adding Unitary Operators}

Now consider the equation with unitary operators $\hat{U}(t)$ and $\hat{U}(t')$:
\begin{equation}
i\hat{\tau}_3 \hat{U}(t_1) \frac{\partial g(t_1, t_2)}{\partial t_1} \hat{U}(t_2) + \hat{U}(t_1) \frac{\partial g(t_1, t_2)}{\partial t_2} \hat{U}(t_2) i\hat{\tau}_3 = 0.
\end{equation}

All objects are matrices.

\paragraph{Goal:} Evolve from $t$ to $t+\Delta t$ in the first argument, holding $t_2 = t'$ fixed.

\paragraph{Step 1: Integrate over time interval.}

\begin{equation}
\int_t^{t+\Delta t} dt_1 \left[i\hat{\tau}_3 \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t_1} \hat{U}(t') + \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t'} \hat{U}(t') i\hat{\tau}_3\right] = 0.
\end{equation}

\paragraph{Step 2: Crank-Nicolson averaging.}

Evaluate the integrand at $t_1 = t$ and $t_1 = t+\Delta t$, then average:

\textbf{At $t_1 = t$:}
\begin{equation}
i\hat{\tau}_3 \hat{U}(t) \frac{\partial g(t, t')}{\partial t_1}\bigg|_{t_1=t} \hat{U}(t') + \hat{U}(t) \frac{\partial g(t, t')}{\partial t'} \hat{U}(t') i\hat{\tau}_3.
\end{equation}

\textbf{At $t_1 = t+\Delta t$:}
\begin{equation}
i\hat{\tau}_3 \hat{U}(t+\Delta t) \frac{\partial g(t+\Delta t, t')}{\partial t_1}\bigg|_{t_1=t+\Delta t} \hat{U}(t') + \hat{U}(t+\Delta t) \frac{\partial g(t+\Delta t, t')}{\partial t'} \hat{U}(t') i\hat{\tau}_3.
\end{equation}

\paragraph{Step 3: Rearrange to isolate derivatives.}

Starting from:
\begin{equation}
i\hat{\tau}_3 \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t_1} \hat{U}(t') + \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t'} \hat{U}(t') i\hat{\tau}_3 = 0.
\end{equation}

Move the second term to the RHS:
\begin{equation}
i\hat{\tau}_3 \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t_1} \hat{U}(t') = -\hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t'} \hat{U}(t') i\hat{\tau}_3.
\end{equation}

Multiply from the left by $\hat{\tau}_3$ (using $\hat{\tau}_3^2 = 1$):
\begin{equation}
\hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t_1} \hat{U}(t') = -\hat{\tau}_3 \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t'} \hat{U}(t') \hat{\tau}_3.
\end{equation}

Multiply from the left by $\hat{U}^{-1}(t_1)$ and from the right by $\hat{U}^{-1}(t')$:
\begin{equation}
\frac{\partial g(t_1, t')}{\partial t_1} = -\hat{U}^{-1}(t_1) \hat{\tau}_3 \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t'} \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t').
\end{equation}

Simplify the right-hand side using $i\hat{\tau}_3 = -\hat{\tau}_3 i$:
\begin{equation}
\frac{\partial g(t_1, t')}{\partial t_1} = \hat{U}^{-1}(t_1) \hat{\tau}_3 \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t'} \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t').
\end{equation}

\paragraph{Step 4: Integrate over $[t, t+\Delta t]$.}

\begin{equation}
\int_t^{t+\Delta t} dt_1 \, \frac{\partial g(t_1, t')}{\partial t_1} = \int_t^{t+\Delta t} dt_1 \, \hat{U}^{-1}(t_1) \hat{\tau}_3 \hat{U}(t_1) \frac{\partial g(t_1, t')}{\partial t'} \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t').
\end{equation}

Left-hand side:
\begin{equation}
\text{LHS} = g(t+\Delta t, t') - g(t, t').
\end{equation}

Right-hand side (Crank-Nicolson): evaluate at $t_1 = t$ and $t_1 = t+\Delta t$, then average and multiply by $\Delta t$:
\begin{equation}
\begin{split}
\text{RHS} &= \frac{\Delta t}{2} \left[\hat{U}^{-1}(t) \hat{\tau}_3 \hat{U}(t) \frac{\partial g(t, t')}{\partial t'} \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t') \right. \\
&\quad \left.  \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) \frac{\partial g(t+\Delta t, t')}{\partial t'} \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t')\right].
\end{split}
\end{equation}

\paragraph{Step 5: Discretize the $\partial_{t'}$ derivatives.}

Using finite differences:
\begin{align}
\frac{\partial g(t, t')}{\partial t'} &\approx \frac{g(t, t'+\Delta t) - g(t, t')}{\Delta t}, \\
\frac{\partial g(t+\Delta t, t')}{\partial t'} &\approx \frac{g(t+\Delta t, t'+\Delta t) - g(t+\Delta t, t')}{\Delta t}.
\end{align}

Substituting:
\begin{equation}
\begin{split}
g(t+\Delta t, t') - g(t, t') &= \frac{1}{2} \Big[\hat{U}^{-1}(t) \hat{\tau}_3 \hat{U}(t) [g(t, t'+\Delta t) - g(t, t')] \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t') \\
&\quad + \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) [g(t+\Delta t, t'+\Delta t) \\
&\quad\quad - g(t+\Delta t, t')] \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t')\Big].
\end{split}
\end{equation}

\paragraph{Step 6: Multiply by unitary operators.}

Multiply the entire equation from the left by $\hat{U}(t)$ and from the right by $\hat{U}(t')$:
\begin{equation}
\begin{split}
\hat{U}(t)[g(t+\Delta t, t') &- g(t, t')]\hat{U}(t') \\
&= \frac{1}{2} \Big[\hat{U}(t) \hat{U}^{-1}(t) \hat{\tau}_3 \hat{U}(t) [g(t, t'+\Delta t) - g(t, t')] \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t') \hat{U}(t') \\
&\quad + \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) [g(t+\Delta t, t'+\Delta t) \\
&\quad\quad - g(t+\Delta t, t')] \hat{U}(t') \hat{\tau}_3 \hat{U}^{-1}(t') \hat{U}(t')\Big].
\end{split}
\end{equation}

Simplify using $\hat{U} \hat{U}^{-1} = 1$:
\begin{equation}
\begin{split}
\hat{U}(t) g(t+\Delta t, t') \hat{U}(t') &- \hat{U}(t) g(t, t') \hat{U}(t') \\
&= \frac{1}{2} \Big[\hat{\tau}_3 \hat{U}(t) [g(t, t'+\Delta t) - g(t, t')] \hat{U}(t') \hat{\tau}_3 \\
&\quad + \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) [g(t+\Delta t, t'+\Delta t) \\
&\quad\quad - g(t+\Delta t, t')] \hat{U}(t') \hat{\tau}_3\Big].
\end{split}
\end{equation}

\paragraph{Step 7: Separate knowns from unknowns.}

Expand the LHS:
\begin{equation}
\hat{U}(t) g(t+\Delta t, t') \hat{U}(t') = \hat{U}(t) g(t, t') \hat{U}(t') + \text{RHS}.
\end{equation}

Expand the RHS terms involving $g(t+\Delta t, \cdot)$:
\begin{equation}
\begin{split}
\text{RHS} &= \frac{1}{2} \hat{\tau}_3 \hat{U}(t) [g(t, t'+\Delta t) - g(t, t')] \hat{U}(t') \hat{\tau}_3 \\
&\quad + \frac{1}{2} \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) g(t+\Delta t, t'+\Delta t) \hat{U}(t') \hat{\tau}_3 \\
&\quad - \frac{1}{2} \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) g(t+\Delta t, t') \hat{U}(t') \hat{\tau}_3.
\end{split}
\end{equation}

Move the last term (containing $g(t+\Delta t, t')$) to the LHS:
\begin{equation}
\begin{split}
\hat{U}(t) g(t+\Delta t, t') \hat{U}(t') &+ \frac{1}{2} \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) g(t+\Delta t, t') \hat{U}(t') \hat{\tau}_3 \\
&= \hat{U}(t) g(t, t') \hat{U}(t') + \frac{1}{2} \hat{\tau}_3 \hat{U}(t) [g(t, t'+\Delta t) - g(t, t')] \hat{U}(t') \hat{\tau}_3 \\
&\quad + \frac{1}{2} \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) g(t+\Delta t, t'+\Delta t) \hat{U}(t') \hat{\tau}_3.
\end{split}
\end{equation}

\paragraph{Step 8: Factor out common terms.}

On the LHS, factor using the identity:
\begin{equation}
A + \frac{1}{2} B \hat{\tau}_3 A \hat{\tau}_3 = \left[1 + \frac{1}{2} B \hat{\tau}_3 \cdot \hat{\tau}_3\right] A,
\end{equation}
where $A = \hat{U}(t) g(t+\Delta t, t') \hat{U}(t')$ and $B = \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t)$.

Actually, this doesn't factor cleanly. Let me write the final form directly:

\textbf{LHS (unknowns at time $t+\Delta t$):}
\begin{equation}
\boxed{
\begin{split}
&\hat{U}(t) g(t+\Delta t, t') \hat{U}(t') + \frac{1}{2} \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) g(t+\Delta t, t') \hat{U}(t') \hat{\tau}_3 \\
&- \frac{1}{2} \hat{U}(t) \hat{U}^{-1}(t+\Delta t) \hat{\tau}_3 \hat{U}(t+\Delta t) g(t+\Delta t, t'+\Delta t) \hat{U}(t') \hat{\tau}_3
\end{split}
}
\end{equation}

\textbf{RHS (knowns at time $t$):}
\begin{equation}
\boxed{
\begin{split}
&\hat{U}(t) g(t, t') \hat{U}(t') + \frac{1}{2} \hat{\tau}_3 \hat{U}(t) [g(t, t'+\Delta t) - g(t, t')] \hat{U}(t') \hat{\tau}_3
\end{split}
}
\end{equation}

\paragraph{Step 9: Simplify unitary operator products.}

Using the composition property $\hat{U}(t+\Delta t) = \hat{U}(\Delta t; t) \hat{U}(t)$, we have:
\begin{align}
\hat{U}^{-1}(t+\Delta t) &= \hat{U}^{-1}(t) \hat{U}^{-1}(\Delta t; t), \\
\hat{U}(t) \hat{U}^{-1}(t+\Delta t) &= \hat{U}(t) \hat{U}^{-1}(t) \hat{U}^{-1}(\Delta t; t) = \hat{U}^{-1}(\Delta t; t).
\end{align}

Substituting into the LHS:
\begin{equation}
\begin{split}
\text{LHS} &= \hat{U}(t) g(t+\Delta t, t') \hat{U}(t') + \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(t+\Delta t) g(t+\Delta t, t') \hat{U}(t') \hat{\tau}_3 \\
&\quad - \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(t+\Delta t) g(t+\Delta t, t'+\Delta t) \hat{U}(t') \hat{\tau}_3.
\end{split}
\end{equation}

Now use $\hat{U}(t+\Delta t) = \hat{U}(\Delta t; t) \hat{U}(t)$:
\begin{equation}
\begin{split}
\text{LHS} &= \hat{U}(t) g(t+\Delta t, t') \hat{U}(t') + \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) \hat{U}(t) g(t+\Delta t, t') \hat{U}(t') \hat{\tau}_3 \\
&\quad - \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) \hat{U}(t) g(t+\Delta t, t'+\Delta t) \hat{U}(t') \hat{\tau}_3.
\end{split}
\end{equation}

Factor out $\hat{U}(t) (\cdots) \hat{U}(t')$ from each term:
\begin{equation}
\begin{split}
\text{LHS} &= \hat{U}(t) \Bigg[g(t+\Delta t, t') + \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) g(t+\Delta t, t') \hat{\tau}_3 \\
&\quad - \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) g(t+\Delta t, t'+\Delta t) \hat{\tau}_3 \Bigg] \hat{U}(t').
\end{split}
\end{equation}

Wait, this doesn't work because the $\hat{\tau}_3$ on the right doesn't commute with $\hat{U}(t')$. Let me reconsider.

Actually, for the RHS we also need to expand similarly. Let me also use $\hat{U}(t') = \hat{U}(\Delta t; t') \hat{U}(t'-\Delta t)$ for the terms involving $\hat{U}(t')$.

But actually, $t'$ is held fixed, so we can't decompose $\hat{U}(t')$ in the same way unless we also consider the evolution in the $t'$ direction.

Let me simplify differently. Define the relative unitary propagator:
\begin{equation}
\hat{U}_R^{-1}(\Delta t; t') \equiv \hat{U}(t') \hat{U}^{-1}(t'+\Delta t) = \hat{U}(t') \hat{U}^{-1}(t') \hat{U}^{-1}(\Delta t; t') = \hat{U}^{-1}(\Delta t; t').
\end{equation}

So the simplified form is:

\textbf{LHS (simplified):}
\begin{equation}
\boxed{
\begin{split}
&\hat{U}(t) g(t+\Delta t, t') \hat{U}(t') \\
&+ \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) \hat{U}(t) g(t+\Delta t, t') \hat{U}(t') \hat{\tau}_3 \\
&- \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) \hat{U}(t) g(t+\Delta t, t'+\Delta t) \hat{U}(t') \hat{\tau}_3
\end{split}
}
\end{equation}

For the term $g(t+\Delta t, t'+\Delta t) \hat{U}(t')$, we can write:
\begin{equation}
g(t+\Delta t, t'+\Delta t) \hat{U}(t') = g(t+\Delta t, t'+\Delta t) \hat{U}(t'+\Delta t) \hat{U}^{-1}(\Delta t; t').
\end{equation}

So:
\begin{equation}
\begin{split}
&\hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) \hat{U}(t) g(t+\Delta t, t'+\Delta t) \hat{U}(t') \hat{\tau}_3 \\
&= \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) \hat{U}(t) g(t+\Delta t, t'+\Delta t) \hat{U}(t'+\Delta t) \hat{U}^{-1}(\Delta t; t') \hat{\tau}_3.
\end{split}
\end{equation}

\textbf{Final simplified form:}

\textbf{LHS:}
\begin{equation}
\boxed{
\begin{split}
&\hat{U}(t) \Bigg[g(t+\Delta t, t') + \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) g(t+\Delta t, t') \hat{\tau}_3 \\
&\quad - \frac{1}{2} \hat{U}^{-1}(\Delta t; t) \hat{\tau}_3 \hat{U}(\Delta t; t) g(t+\Delta t, t'+\Delta t) \hat{U}^{-1}(\Delta t; t') \hat{\tau}_3\Bigg] \hat{U}(t').
\end{split}
}
\end{equation}

\textbf{RHS:}
\begin{equation}
\boxed{
\hat{U}(t) \left[g(t, t') + \frac{1}{2} \hat{\tau}_3 [g(t, t'+\Delta t) - g(t, t')] \hat{\tau}_3\right] \hat{U}(t').
}
\end{equation}
